示例：

```
定义：
action chunk长度为h_a=16
动作维度：d_a=13
单步快速观测维度：d_f=6
h_o=2
chunk内实际执行步数：h_e=6
延迟长度：l=2

慢循环：
while 未收敛 do
	获取慢速观测：
		O_slow <- GetSlowObservation(length=h_o)
		O_slow = {I_slow, P_slow, tcp_wrench}，包含过去h_o步的 视觉图像I_t + 本体感知P_t
            I_slow形状为[batch, h_o, 3, 360, 640]
            P_slow形状为[batch, h_o, 6]
            tcp_wrench形状为[batch, h_o, 6]
	编码慢速上下文并缓存：
		Z_slow <- E_slow(O_slow)
		Z_slow形状为[batch, slow token数=102, 768]
		E_slow = 图像编码分支 + 低维状态编码分支 + token 拼接
            图像分支:
                I_feat <- ResNet34(I_slow)
                I_tokens <- FlattenSpatial(I_feat)
                I_tokens(98) <- LinearProjection(I_tokens)
            低维分支:
                P_tokens(2) <- LinearProjection(P_slow)
                W_tokens(2) <- LinearProjection(W_slow)
            输出:
                Z_slow(102个token) <- Concat(I_tokens, P_tokens, W_tokens)
		
	采样高斯噪声序列：A^K <- N(0, I)，长度为h_a，形状为[batch, h_a, d_a]
		满足h_a >= h_o+h_e-1+l
	
	快循环
	for i = 0 to h_e-1 do
		获取快速力觉观测：
			O_fast <- GetFastObservation(length=h_o+i+l)
            O_fast的形状为[batch, h_o+i+l, d_f=6]
            h_o对应历史观测，l为延时补偿长度，(h_o+l)为序列的基础长度
			
		编码快速上下文:
            Z_fast <- E_fast(O_fast)
            E_fast：GRU+线性层
            Z_fast形状为[batch, h_o+i+l, token embedding维度 768]
            
        取当前长度对应的噪声前缀:
            A_prefix^K <- Prefix(A^K, h_o+i+l)
            A_prefix^K形状为[batch, h_o+i+l, d_a]
        使用确定性 DDIM 去噪（eta = 0）:
            A_prefix_hat^0 <- DDIM(pi_theta, Z_slow, Z_fast, A_prefix^K, eta = 0)
            pi_theta为训练好的去噪网络
            
            具体执行流程：
            trajectory <- A_prefix^K
            for k = K, K-1, ..., 1 do
                用 pi_theta 预测当前 trajectory(A_prefix^k) 在第 k 步的velocity预测
                	 v_prediction <- pi_theta(trajectory, k, Z_slow, Z_fast)
                	（pi_theta具体见下面的拆解分析）
                根据 DDIM 更新公式，把 trajectory 从 k 步推进到 k-1 步
                	trajectory <- DDIMStep(trajectory, v_prediction， k)
                	具体流程：
                		由 velocity 还原 x_hat_0 和 epsilon_hat：
                		x_hat_0 <- sqrt(alpha_bar_k) * trajectory 
                						- sqrt(1 - alpha_bar_k) * v_prediction
                        epsilon_hat <- sqrt(alpha_bar_k) * v_prediction
                                    	+ sqrt(1 - alpha_bar_k) * trajectory
                        在 eta=0 条件下计算 x_{t-1}：
                        trajectory <- sqrt(alpha_bar_{k-1}) * x_hat_0
                                      + sqrt(1 - alpha_bar_{k-1}) * epsilon_hat

            end for
            A_prefix_hat^0 <- trajectory
            
        取最后一个预测动作:
            a_hat,t <- LastAction(A_prefix_hat^0)
	
		丢弃辅助部分，仅保留真实动作:
            a_real,t <- RealPart(a_hat,t)
		执行真实动作:
            Execute(a_real,t)
	end for
end for
```
